+                .PostOperation()

+                .WithDescription("Sends the new account (id, name) to the ERP.")
+                .Handle(OnExecute);
+        }
+

+            => services.AddScoped<IErpAccountSender>(r => new ErpAccountSender(SecureConfig));
+

+        {
+            ctx.Trace("AccountErpNotify: sending {0} '{1}' to the ERP.", ctx.PrimaryEntityName, ctx.PrimaryEntityId);
+
+            if (string.IsNullOrWhiteSpace(SecureConfig))
+            {
+                throw new InvalidPluginExecutionException(
+                    "The ERP API key is not configured: set the step's secure configuration 'ErpApiKey'.");
+            }
+

+            ctx.Trace("AccountErpNotify: ERP accepted account '{0}'.", ctx.PrimaryEntityId);
+        }

+namespace Zava.Integrations.Account.Services
+{
+    using System;
+    using System.IO;
+    using System.Net.Http;
+    using System.Runtime.Serialization;
+    using System.Runtime.Serialization.Json;
+    using System.Text;
+    using System.Threading.Tasks;
+    using Microsoft.Xrm.Sdk;
+
+    /// <summary>
+    /// Sends a newly created account to the ERP. Generated by <c>vela add service</c>; registered from
+    /// <c>AccountErpNotify.ConfigureServices(...)</c> and resolved with <c>ctx.GetRequiredService&lt;IErpAccountSender&gt;()</c>.
+    /// </summary>
+    public interface IErpAccountSender
+    {
+        void Send(Guid accountId, string name);
+    }
+
+    /// <summary>
+    /// POSTs <c>{"id":"&lt;guid&gt;","name":"&lt;name&gt;"}</c> to the ERP with the API key in <c>X-Api-Key</c>.
+    /// Any non-2xx answer, transport error or timeout throws <see cref="InvalidPluginExecutionException"/>, which fails
+    /// the async system job (no retry: ops replays it).
+    /// </summary>
+    public sealed class ErpAccountSender : IErpAccountSender
+    {
+        public const string Endpoint = "https://erp.zava-demo.com/api/accounts";
+        public const string ApiKeyHeader = "X-Api-Key";
+
+        /// <summary>Explicit timeout, well under the 2-minute sandbox limit.</summary>
+        public static readonly TimeSpan DefaultTimeout = TimeSpan.FromSeconds(30);
+
+        private readonly string _apiKey;
+        private readonly HttpMessageHandler _handler;
+        private readonly TimeSpan _timeout;
+
+        /// <summary>Production: the default transport (owned and disposed per call) and <see cref="DefaultTimeout"/>.</summary>
+        public ErpAccountSender(string apiKey)
+        {
+            _apiKey = apiKey;
+            _handler = null;
+            _timeout = DefaultTimeout;
+        }
+
+        /// <summary>A caller-owned transport (e.g. a fake in tests) and an explicit timeout.</summary>
+        public ErpAccountSender(string apiKey, HttpMessageHandler handler, TimeSpan timeout)
+        {
+            _apiKey = apiKey;
+            _handler = handler ?? throw new ArgumentNullException(nameof(handler));
+            _timeout = timeout;
+        }
+
+        public void Send(Guid accountId, string name)
+        {
+            string json = Serialize(new ErpAccountPayload { Id = accountId.ToString("D"), Name = name });
+
+            using (HttpClient client = _handler == null ? new HttpClient() : new HttpClient(_handler, disposeHandler: false))
+            using (var request = new HttpRequestMessage(HttpMethod.Post, Endpoint))
+            {
+                client.Timeout = _timeout;
+                request.Headers.Add(ApiKeyHeader, _apiKey);
+                request.Content = new StringContent(json, Encoding.UTF8, "application/json");
+
+                HttpResponseMessage response;
+                try
+                {
+                    response = client.SendAsync(request).GetAwaiter().GetResult();
+                }
+                catch (TaskCanceledException)
+                {
+                    throw new InvalidPluginExecutionException(
+                        $"ERP call for account {accountId} timed out after {_timeout.TotalSeconds} s.");
+                }
+                catch (HttpRequestException ex)
+                {
+                    throw new InvalidPluginExecutionException(
+                        $"ERP unreachable for account {accountId}: {ex.Message}", ex);
+                }
+
+                using (response)
+                {
+                    if (!response.IsSuccessStatusCode)
+                    {
+                        throw new InvalidPluginExecutionException(
+                            $"ERP rejected account {accountId}: HTTP {(int)response.StatusCode} ({response.ReasonPhrase}).");
+                    }
+                }
+            }
+        }
+
+        private static string Serialize(ErpAccountPayload payload)
+        {
+            var serializer = new DataContractJsonSerializer(typeof(ErpAccountPayload));
+            using (var stream = new MemoryStream())
+            {
+                serializer.WriteObject(stream, payload);
+                return Encoding.UTF8.GetString(stream.ToArray());
+            }
+        }
+    }
+
+    /// <summary>The ERP request body. Public: the sandbox's partial trust only serializes public types.</summary>
+    [DataContract]
+    public sealed class ErpAccountPayload
+    {
+        [DataMember(Name = "id", Order = 0)]
+        public string Id { get; set; }
+
+        [DataMember(Name = "name", Order = 1)]
+        public string Name { get; set; }
+    }
